Replace inside AdminConsole.jsx:

const events = useAdminStream();

Then replace AI Activity Stream block with:

<div style={styles.stream}>
  <h3 style={styles.sectionTitle}>⚡ AI Activity Stream (LIVE)</h3>

  {events.length === 0 && (
    <div style={{ opacity: 0.6 }}>Waiting for system events...</div>
  )}

  {events.map((e, i) => (
    <div key={i} style={styles.logLine}>
      <span style={{ color: "#00F5FF" }}>{e.type}</span>
      <span style={{ marginLeft: 10 }}>
        {JSON.stringify(e.payload)}
      </span>
    </div>
  ))}
</div>
